// src/Plugin/Field/FieldWidget/CustomImageFieldWidget.php

namespace Drupal\ecommerce\Plugin\Field\FieldWidget;

use Drupal\Core\Field\FieldItemListInterface;
use Drupal\Core\Field\WidgetBase;
use Drupal\Core\Form\FormStateInterface;

/**
 * @FieldWidget(
 *   id = "custom_image_field_widget",
 *   label = @Translation("Custom Image Field Widget"),
 *   field_types = { "custom_image_field" }
 * )
 */
class CustomImageFieldWidget extends WidgetBase {
  public function formElement(FieldItemListInterface $items, $delta, array $element, array &$form, FormStateInterface $form_state) {
    // Defines input fields for text and URL based on 0.5.10
    $element['alt'] = ['#type' => 'textfield', 
    '#title' => $this->t('Alternative Text'), '#default_value' => $items[$delta]->alt ?? ''];
    $element['uri'] = ['#type' => 'textarea', 
    '#title' => $this->t('Image URI'), '#default_value' => $items[$delta]->uri ?? ''];
    return $element;
  }
}
